enable
configure terminal
hostname F1-router

! --- WAN Interfaces (Point-to-Point) ---
interface s0/2/1
 description Connection to R2 (Floor 2)
 ip address 10.10.10.9 255.255.255.252
 no shutdown
exit

interface s0/2/0
 description Connection to R3 (Floor 3)
 ip address 10.10.10.5 255.255.255.252
 no shutdown
exit

! --- VLAN 80: Reception Subinterface & DHCP ---
interface f0/0
 no shutdown
exit

interface f0/0.80
 encapsulation dot1Q 80
 ip address 192.168.8.1 255.255.255.0
exit

ip dhcp pool Reception
 network 192.168.8.0 255.255.255.0
 default-router 192.168.8.1
 dns-server 8.8.8.8
exit


! --- VLAN 70: Store Subinterface & DHCP ---
interface f0/0.70
 encapsulation dot1Q 70
 ip address 192.168.7.1 255.255.255.0
exit

ip dhcp pool Store
 network 192.168.7.0 255.255.255.0
 default-router 192.168.7.1
 dns-server 8.8.8.8
exit


! --- VLAN 60: Logistics Subinterface & DHCP ---
interface f0/0.60
 encapsulation dot1Q 60
 ip address 192.168.6.1 255.255.255.0
exit

ip dhcp pool Logistics
 network 192.168.6.0 255.255.255.0
 default-router 192.168.6.1
 dns-server 8.8.8.8
exit


! --- OSPF Dynamic Routing ---
router ospf 10
 network 10.10.10.0 0.0.0.3 area 0
 network 10.10.10.8 0.0.0.3 area 0
 network 192.168.8.0 0.0.0.255 area 0
 network 192.168.7.0 0.0.0.255 area 0
 network 192.168.6.0 0.0.0.255 area 0
exit

! --- Security & User Setup for SSH ---
ip domain-name Akshay
crypto key generate rsa general-keys modulus 1024
username Akshay password <password>

line vty 0 15
 transport input ssh
 login local
exit
